cmake_minimum_required(VERSION 3.22)
set(IDF_TARGET esp32p4)
set(COMPONENTS main)
set(EXTRA_COMPONENT_DIRS
    "${CMAKE_CURRENT_LIST_DIR}/../../components/libjpeg-turbo-esp32"
    "${CMAKE_CURRENT_LIST_DIR}/../../components/libwebp-esp32"
    "${CMAKE_CURRENT_LIST_DIR}/home_assistant")
set(PROJECT_VER "swipe-prototype")
set(SWIPE_BOARD "v3" CACHE STRING "Guition JC8012P4A1 prototype board: v1, v2, or v3")
option(SWIPE_TRACE "Record native playback timings over USB" OFF)
option(SWIPE_ESPHOME_TRIAL "Build a timed V1 swipe trial for the ESPHome app slot" OFF)
option(SWIPE_TRIAL_IMMICH "Enable live Immich playback in the V1 ESPHome app-slot trial" OFF)
option(SWIPE_TRIAL_BENCHMARK "Use scripted swipes in the live V1 trial" OFF)
option(SWIPE_TRIAL_BENCHMARK_RAPID "Allow rapid scripted swipes to reach a cold cache edge" OFF)
option(SWIPE_TRIAL_BENCHMARK_FIXTURES "Use local slides in the scripted V1 benchmark" OFF)
set(SWIPE_TRIAL_REPLAY_ORIGIN "" CACHE STRING "Local synthetic Immich HTTP origin for the V1 rapid trial")
option(SWIPE_TRIAL_REPLAY_PREPARED "Fetch prepared portrait pairs during the replay trial" OFF)
option(SWIPE_TRIAL_REPLAY_HISTORY "Rewind through prepared replay history during an outage" OFF)
option(SWIPE_TRIAL_REPLAY_DISABLE_LOOKAHEAD "Use sequential preview downloads for the direct replay control" OFF)
option(SWIPE_TRIAL_REPLAY_DISABLE_COMPANION_PREFETCH "Skip companion prefetch while retaining primary lookahead" OFF)
set(SWIPE_NATIVE_OTA_TEST "OFF" CACHE STRING "Native OTA recovery test image: OFF, BASELINE, or FAIL")
if(NOT SWIPE_BOARD MATCHES "^v[123]$")
    message(FATAL_ERROR "SWIPE_BOARD must be v1, v2, or v3")
endif()
if(SWIPE_TRIAL_IMMICH AND NOT SWIPE_ESPHOME_TRIAL)
    message(FATAL_ERROR "SWIPE_TRIAL_IMMICH requires SWIPE_ESPHOME_TRIAL")
endif()
if(SWIPE_TRIAL_BENCHMARK AND NOT SWIPE_TRIAL_IMMICH)
    message(FATAL_ERROR "SWIPE_TRIAL_BENCHMARK requires SWIPE_TRIAL_IMMICH")
endif()
if(SWIPE_TRIAL_BENCHMARK_RAPID AND NOT SWIPE_TRIAL_BENCHMARK)
    message(FATAL_ERROR "SWIPE_TRIAL_BENCHMARK_RAPID requires SWIPE_TRIAL_BENCHMARK")
endif()
if(SWIPE_TRIAL_BENCHMARK_FIXTURES AND NOT SWIPE_TRIAL_BENCHMARK)
    message(FATAL_ERROR "SWIPE_TRIAL_BENCHMARK_FIXTURES requires SWIPE_TRIAL_BENCHMARK")
endif()
if(SWIPE_TRIAL_REPLAY_ORIGIN AND
   (NOT SWIPE_TRIAL_REPLAY_ORIGIN MATCHES "^http://[0-9.]+:[0-9]+$" OR
    NOT SWIPE_TRIAL_BENCHMARK_RAPID OR SWIPE_TRIAL_BENCHMARK_FIXTURES))
    message(FATAL_ERROR "Replay needs a local HTTP IPv4 origin and the rapid V1 trial")
endif()
if(SWIPE_TRIAL_REPLAY_PREPARED AND NOT SWIPE_TRIAL_REPLAY_ORIGIN)
    message(FATAL_ERROR "Prepared replay requires SWIPE_TRIAL_REPLAY_ORIGIN")
endif()
if(SWIPE_TRIAL_REPLAY_HISTORY AND NOT SWIPE_TRIAL_REPLAY_PREPARED)
    message(FATAL_ERROR "History replay requires SWIPE_TRIAL_REPLAY_PREPARED")
endif()
if(SWIPE_TRIAL_REPLAY_DISABLE_LOOKAHEAD AND (NOT SWIPE_TRIAL_REPLAY_ORIGIN OR SWIPE_TRIAL_REPLAY_PREPARED))
    message(FATAL_ERROR "The lookahead control requires a direct synthetic replay")
endif()
if(SWIPE_TRIAL_REPLAY_DISABLE_COMPANION_PREFETCH AND
   (NOT SWIPE_TRIAL_REPLAY_ORIGIN OR SWIPE_TRIAL_REPLAY_PREPARED OR SWIPE_TRIAL_REPLAY_DISABLE_LOOKAHEAD))
    message(FATAL_ERROR "The companion prefetch control requires a direct synthetic replay with lookahead enabled")
endif()
if(NOT SWIPE_NATIVE_OTA_TEST MATCHES "^(OFF|BASELINE|FAIL)$")
    message(FATAL_ERROR "SWIPE_NATIVE_OTA_TEST must be OFF, BASELINE, or FAIL")
endif()
if(SWIPE_ESPHOME_TRIAL AND NOT SWIPE_NATIVE_OTA_TEST STREQUAL "OFF")
    message(FATAL_ERROR "Native OTA test images need the native partition layout")
endif()
set(SDKCONFIG "${CMAKE_BINARY_DIR}/sdkconfig")
set(SDKCONFIG_DEFAULTS "${CMAKE_CURRENT_LIST_DIR}/sdkconfig.defaults")
if(NOT SWIPE_BOARD STREQUAL "v3")
    list(APPEND SDKCONFIG_DEFAULTS "${CMAKE_CURRENT_LIST_DIR}/sdkconfig.early-p4")
endif()
set(espframe_project "espframe_native_${SWIPE_BOARD}")
set(espframe_partitions "partitions.csv")
if(SWIPE_ESPHOME_TRIAL)
    if(NOT SWIPE_BOARD STREQUAL "v1")
        message(FATAL_ERROR "The ESPHome app-slot trial only supports the original Guition V1")
    endif()
    if(SWIPE_TRIAL_IMMICH)
        list(APPEND SDKCONFIG_DEFAULTS "${CMAKE_CURRENT_LIST_DIR}/sdkconfig.esphome-live-trial")
        if(SWIPE_TRIAL_BENCHMARK_FIXTURES)
            set(espframe_project "espframe_v1_fixture_benchmark")
        elseif(SWIPE_TRIAL_BENCHMARK_RAPID)
            set(espframe_project "espframe_v1_rapid_benchmark")
        elseif(SWIPE_TRIAL_BENCHMARK)
            set(espframe_project "espframe_v1_swipe_benchmark")
        else()
            set(espframe_project "espframe_v1_live_trial")
        endif()
    else()
        list(APPEND SDKCONFIG_DEFAULTS "${CMAKE_CURRENT_LIST_DIR}/sdkconfig.esphome-trial")
        set(espframe_project "espframe_v1_legacy_swipe_trial")
    endif()
    set(espframe_partitions "partitions.esphome-v1.csv")
else()
    list(APPEND SDKCONFIG_DEFAULTS "${CMAKE_CURRENT_LIST_DIR}/sdkconfig.native")
    if(NOT SWIPE_NATIVE_OTA_TEST STREQUAL "OFF")
        set(espframe_project "espframe_ota_test_${SWIPE_BOARD}")
        set(PROJECT_VER "native-ota-test-${SWIPE_NATIVE_OTA_TEST}")
    endif()
endif()
include($ENV{IDF_PATH}/tools/cmake/project.cmake)
project(${espframe_project})
if(NOT CONFIG_PARTITION_TABLE_CUSTOM_FILENAME STREQUAL espframe_partitions)
    message(FATAL_ERROR "Use a separate build directory for native installs and the ESPHome app-slot trial")
endif()
if((SWIPE_TRIAL_IMMICH AND NOT CONFIG_ESP_HOSTED) OR
   (SWIPE_ESPHOME_TRIAL AND NOT SWIPE_TRIAL_IMMICH AND CONFIG_ESP_HOSTED))
    message(FATAL_ERROR "Use separate build directories for the live and offline V1 trial.")
endif()
