cmake_minimum_required(VERSION 3.22)
set(IDF_TARGET esp32p4)
set(COMPONENTS main)
set(EXTRA_COMPONENT_DIRS
    "${CMAKE_CURRENT_LIST_DIR}/../../components/libjpeg-turbo-esp32"
    "${CMAKE_CURRENT_LIST_DIR}/../../components/libwebp-esp32")
set(PROJECT_VER "swipe-prototype")
set(SWIPE_BOARD "v3" CACHE STRING "Guition JC8012P4A1 prototype board: v1, v2, or v3")
option(SWIPE_TRACE "Record native playback timings over USB" OFF)
option(SWIPE_ESPHOME_TRIAL "Build a timed V1 swipe trial for the ESPHome app slot" OFF)
if(NOT SWIPE_BOARD MATCHES "^v[123]$")
    message(FATAL_ERROR "SWIPE_BOARD must be v1, v2, or v3")
endif()
set(SDKCONFIG "${CMAKE_BINARY_DIR}/sdkconfig")
set(SDKCONFIG_DEFAULTS "${CMAKE_CURRENT_LIST_DIR}/sdkconfig.defaults")
if(NOT SWIPE_BOARD STREQUAL "v3")
    list(APPEND SDKCONFIG_DEFAULTS "${CMAKE_CURRENT_LIST_DIR}/sdkconfig.early-p4")
endif()
set(espframe_project "espframe_native_${SWIPE_BOARD}")
set(espframe_partitions "partitions.csv")
if(SWIPE_ESPHOME_TRIAL)
    if(NOT SWIPE_BOARD STREQUAL "v1")
        message(FATAL_ERROR "The ESPHome app-slot trial only supports the original Guition V1")
    endif()
    list(APPEND SDKCONFIG_DEFAULTS "${CMAKE_CURRENT_LIST_DIR}/sdkconfig.esphome-trial")
    set(espframe_project "espframe_v1_legacy_swipe_trial")
    set(espframe_partitions "partitions.esphome-v1.csv")
endif()
include($ENV{IDF_PATH}/tools/cmake/project.cmake)
project(${espframe_project})
if(NOT CONFIG_PARTITION_TABLE_CUSTOM_FILENAME STREQUAL espframe_partitions)
    message(FATAL_ERROR "Use a separate build directory for native installs and the ESPHome app-slot trial")
endif()
