cmake_minimum_required(VERSION 3.22)

if(NOT ESPFRAME_HA_HOST)
    set(IDF_TARGET esp32p4)
    set(COMPONENTS main)
    set(SDKCONFIG "${CMAKE_BINARY_DIR}/sdkconfig")
    set(EXTRA_COMPONENT_DIRS "${CMAKE_CURRENT_LIST_DIR}/../../home_assistant")
    include($ENV{IDF_PATH}/tools/cmake/project.cmake)
    project(espframe_ha_link_test)
else()
    project(espframe_ha_host_test LANGUAGES C CXX)
    find_package(Threads REQUIRED)
    add_subdirectory("${ESPHOME_MANAGED_COMPONENTS}/esphome__libsodium" sodium)
    add_subdirectory("${ESPHOME_MANAGED_COMPONENTS}/esphome__noise-c" noise)
    target_compile_definitions(noise_c PRIVATE
        NOISE_USE_PROTOCOL_NAME_TABLE=0 NOISE_USE_FALLBACK=0 NOISE_USE_HFS=0)
    if(APPLE)
        add_link_options(-Wl,-dead_strip)
    else()
        target_compile_options(sodium PRIVATE -ffunction-sections -fdata-sections)
        add_link_options(-Wl,--gc-sections)
    endif()

    add_executable(espframe_ha_host_test host_server.cpp ../../home_assistant/esphome_session.cpp
                   ../../home_assistant/esphome_socket.cpp)
    target_compile_features(espframe_ha_host_test PRIVATE cxx_std_20)
    target_compile_options(espframe_ha_host_test PRIVATE -Wall -Wextra -Werror)
    target_include_directories(espframe_ha_host_test PRIVATE ../../main ../../home_assistant ../../../../components/espframe)
    target_link_libraries(espframe_ha_host_test PRIVATE noise_c Threads::Threads)
endif()
