cmake_minimum_required(VERSION 3.16)
project(photo_buffer_lvgl_tests LANGUAGES C CXX)
if(NOT LVGL_SOURCE)
  message(FATAL_ERROR "Set LVGL_SOURCE to the LVGL source from a firmware build")
endif()
set(LV_BUILD_CONF_PATH "${CMAKE_CURRENT_SOURCE_DIR}/lv_conf.h" CACHE PATH "" FORCE)
set(CONFIG_LV_BUILD_DEMOS OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_USE_THORVG_INTERNAL OFF CACHE BOOL "" FORCE)
add_subdirectory("${LVGL_SOURCE}" lvgl)
add_executable(photo_buffer_lvgl_tests test.cpp)
target_compile_features(photo_buffer_lvgl_tests PRIVATE cxx_std_17)
target_include_directories(photo_buffer_lvgl_tests PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/../..")
target_link_libraries(photo_buffer_lvgl_tests PRIVATE lvgl::lvgl m)
